      private function testBullet(w:World,id:String,x:Number,y:Number):Bullet {
         var weapon:Object=w.invent.weapons[id];var b:Bullet=new Bullet(w.gg,x,y,weapon.vBullet);
         b.weap=weapon;b.damage=10;b.tipDamage=weapon.tipDamage;b.vel=10;b.dx=10;b.knockx=1;b.knocky=0;b.liv=40;return b;
      }
      private function clearTestEffects(target:Object):void {
         var list:Array=target.effects.concat();for each(var effect:Object in list)if(!effect.vse)effect.unsetEff(false,false,false);
         target.effects=[];target.setEffParams();
      }
